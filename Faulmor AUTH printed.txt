local function _transport(url, method, body, headers)
    local _req = nil
    if type(_G) == 'table' and type(_G.request) == 'function' then _req = _G.request end
    if not _req and type(syn) == 'table' and type(syn.request) == 'function' then _req = syn.request end
    if not _req and type(http) == 'table' and type(http.request) == 'function' then _req = http.request end
    if not _req and type(fluxus) == 'table' and type(fluxus.request) == 'function' then _req = fluxus.request end
    if _req then
        warn('[fm] DIAG S2: transport via request() family')
        local success, response = pcall(_req, {
            Url = url,
            Method = method,
            Headers = headers or {['Content-Type'] = 'application/json'},
            Body = body
        })
        if not success then
            warn('[fm] DIAG S2: request() error: ' .. tostring(response))
        elseif type(response) == 'table' then
            local sc = response.StatusCode or response.Status or response.status_code or 0
            local b = response.Body or response.body or response.Data or response.data
            warn('[fm] DIAG S2: request() returned status=' .. tostring(sc) .. ' body_type=' .. type(b))
            if b ~= nil then
                return tostring(b or ''), true, tonumber(sc) or 0
            end
            warn('[fm] DIAG S2: request() table had no Body — falling through')
        else
            warn('[fm] DIAG S2: request() odd response type=' .. type(response))
        end
    else
        warn('[fm] DIAG S2: no request() family, trying HttpService')
    end

    if hs and type(hs.PostAsync) == 'function' and type(hs.GetAsync) == 'function' then
        warn('[fm] DIAG S2: transport via HttpService')
        if method == 'POST' then
            local success, response = pcall(hs.PostAsync, hs, url, body, Enum.HttpContentType.ApplicationJson, false)
            if success then return tostring(response or ''), true, 200 end
            warn('[fm] DIAG S2: HttpService PostAsync failed: ' .. tostring(response))
            return tostring(response), false, 500
        else
            local success, response = pcall(hs.GetAsync, hs, url, true)
            if success then return tostring(response or ''), true, 200 end
            warn('[fm] DIAG S2: HttpService GetAsync failed: ' .. tostring(response))
            return tostring(response), false, 500
        end
    end
    warn('[fm] DIAG S2: no usable transport available')
    return '', false, 0
end
